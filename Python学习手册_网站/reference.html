一、安装与工程基础

1.1 pip —— 包管理器

bash
pip install requests                # 安装
pip install requests==2.32.3        # 装指定版本
pip install "requests>=2.30"        # 装某个范围
pip install -U requests             # 升级
pip uninstall requests              # 卸载
pip list                            # 看装了什么
pip show requests                   # 看某个包的详情

1.2 虚拟环境 —— 每个项目一个，非常重要

bash
python -m venv .venv                        # 创建
.venv\Scripts\activate                       # Windows 激活
source .venv/bin/activate                   # macOS / Linux
deactivate                                  # 退出
⚠️ 为什么要用：不同项目依赖的版本可能冲突（A 项目要 numpy 1.x，B 项目要 2.x）。
全局装会互相打架，虚拟环境让每个项目各玩各的。
VS Code 里按 Ctrl+Shift+P → Python: Select Interpreter → 选 .venv。

python
from collections import Counter, defaultdict, deque
from heapq import heappush, heappop
from bisect import bisect_left, insort
from itertools import permutations, combinations, product
from functools import lru_cache
import math

# ---- 计数 ----
print(Counter("abracadabra").most_common(2))     # [('a', 5), ('b', 2)]

# ---- 缺键自动初始化 ----
graph = defaultdict(list)
graph["A"].append("B")                            # 不用先判断 A 在不在

# ---- 队列 ----
q = deque([1, 2, 3]); q.appendleft(0); print(q.popleft())   # 0

# ---- 优先队列（最小堆） ----
h = []; heappush(h, 5); heappush(h, 1); print(heappop(h))   # 1
# 最大堆技巧：存负数

# ---- 二分 ----
arr = [1, 3, 5, 7]
print(bisect_left(arr, 5))                        # 2
insort(arr, 4); print(arr)                        # [1,3,4,5,7]

# ---- 组合 / 排列 ----
print(list(combinations([1,2,3], 2)))             # [(1,2),(1,3),(2,3)]
print(list(permutations([1,2,3], 2))[:2])         # [(1,2),(1,3)]

# ---- 记忆化递归（斐波那契） ----
@lru_cache(maxsize=None)
def fib(n): return n if n < 2 else fib(n-1) + fib(n-2)
print(fib(100))                                   # 秒出

print(math.gcd(12, 18), math.isqrt(17), math.comb(5, 2))
💡 算法题的心法：先想清楚"数据规模"决定复杂度。
n ≤ 20 可以指数级暴搜；n ≤ 5000 用 O(n²)；n ≤ 10⁵~10⁶ 必须 O(n log n) 或 O(n)。
本作业里的"查重"用 set 就是 O(n)，如果用双重循环就是 O(n²)——这就是选对容器的价值。

        tasks = [client.get("https://httpbin.org/delay/1") for _ in range(5)]

class BooksSpider(scrapy.Spider):
    name = "books"
    start_urls = ["https://books.toscrape.com/"]

import socket

# ---- 端口连通性检查（只查自己的服务）----
def check(host, port, timeout=2):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.settimeout(timeout)
        return s.connect_ex((host, port)) == 0

print(check("127.0.0.1", 80))

# ---- 一个只监听本机的迷你 TCP 服务端 ----
# server.py
import socket
srv = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
srv.bind(("127.0.0.1", 9999)); srv.listen(5)
print("listening on 127.0.0.1:9999")
while True:
    conn, addr = srv.accept()
    with conn:
        print("来自", addr, ":", conn.recv(1024).decode(errors="ignore"))
        conn.sendall(b"hello from server")

import hashlib, hmac, secrets

# ---- 摘要：文件校验 ----
print(hashlib.sha256(b"hello").hexdigest())
def file_sha256(path, chunk=8192):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while (b := f.read(chunk)):
            h.update(b)
    return h.hexdigest()

# ---- HMAC：验证消息没被改过 ----
key = b"my-secret"
sig = hmac.new(key, b"important data", hashlib.sha256).hexdigest()

⚠️ 重要常识

  • 别用 MD5/SHA1 存密码（可被撞库/彩虹表）。存密码用 bcrypt / argon2，它们自带"慢"和"加盐"

  • 哈希 ≠ 加密：哈希不可逆，加密可解密，用途完全不同

# ---- 一个最小神经网络（手写数字分类）----
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

tf = transforms.ToTensor()
train_ds = datasets.MNIST("./data", train=True, download=True, transform=tf)
train_dl = DataLoader(train_ds, batch_size=64, shuffle=True)

net = nn.Sequential(nn.Flatten(), nn.Linear(28*28, 128), nn.ReLU(), nn.Linear(128, 10))
loss_fn = nn.CrossEntropyLoss()
opt = optim.Adam(net.parameters(), lr=1e-3)

net.train()
for epoch in range(1):
    for xb, yb in train_dl:
        opt.zero_grad()                  # ① 清梯度
        loss = loss_fn(net(xb), yb)      # ② 前向 + 算损失
        loss.backward()                  # ③ 反向传播
        opt.step()                       # ④ 更新参数
    print("epoch", epoch, "loss", loss.item())

备注

  • ✅ 训练循环死记四步：zero_grad → 前向+loss → backward → step
  • ⚠️ 忘了 zero_grad()，梯度会累加，模型学不对
  • ⚠️ 训练时 net.train()，推理时 net.eval() + with torch.no_grad():
  • ✅ 有 NVIDIA 显卡装 CUDA 版才快（pip install torch --index-url https://download.pytorch.org/whl/cu121）

        level = "优秀"

    return {"姓名": stu.姓名, "等级": level}

sqlite3 —— 标准库自带的小型数据库

定位：单文件数据库，零安装、零配置，脚本和小项目首选。

用途：一个文件就是一个数据库：持久化保存爬到的数据、做本地工具 / 原型的存储（零安装零配置）

安装：内置，import sqlite3

python
import sqlite3

conn = sqlite3.connect("school.db")
cur = conn.cursor()
cur.execute("""CREATE TABLE IF NOT EXISTS student(
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL, score INTEGER)""")

# ⚠️ 一定要用占位符 ?，不要用 f-string 拼 SQL（会被 SQL 注入）
cur.executemany("INSERT INTO student(name, score) VALUES (?, ?)",
                [("张三", 686), ("李四", 655), ("王五", 700)])
conn.commit()

cur.execute("SELECT name, score FROM student WHERE score >= ? ORDER BY score DESC", (660,))
for row in cur.fetchall():
    print(row)

cur.execute("SELECT AVG(score) FROM student")
print("平均分:", cur.fetchone()[0])

cur.execute("UPDATE student SET score = ? WHERE name = ?", (690, "李四"))
conn.commit()
conn.close()
